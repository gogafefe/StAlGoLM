148 // Страховка: если производная всё же оказалась близка к нулю
149 // (не должно случаться при нашем подборе x0, но лучше проверить),
150 // сдвигаем x0 на небольшой шаг и пересчитываем.

160 // Типичные ошибки при подстановке в формулу Ньютона:
161 double wrong1 = x0 + fx0 / fpx0; // перепутан знак
162 double wrong3 = fx0 / fpx0; // забыли вычесть из x0

169 }

172
173 // Гарантируем, что все 4 варианта визуально различимы после округления до 4 знаков
174 for (int i = 1; i < values.size(); i++) {
175 for (int j = 0; j < i; j++) {
176 while (std::fabs(values[i] - values[j]) < 0.01) {
177 values[i] += 0.37 + i;
178 }
179 }
180 }

185 for (int i = order.size() - 1; i > 0; i--) {
186 int j = rng->bounded(0, i + 1);
187 std::swap(order[i], order[j]);
188 }

197 }
