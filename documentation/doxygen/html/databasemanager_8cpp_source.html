74void DatabaseManager::initStatisticTables()
75{
76 QSqlQuery query;
77
78 // Генерируем строку с 40 столбцами p1..p40
79 QString columns;
80 for (int i = 1; i <= 40; i++) {
81 if (i > 1) columns += ", ";
82 columns += QString("p%1 INTEGER DEFAULT 0").arg(i);
83 }
84
85 // Создаём 4 таблицы статистики для Task1..Task4
86 for (int taskNum = 1; taskNum <= 4; taskNum++) {
87 QString tableName = QString("StatisticTask%1").arg(taskNum);
88 QString sql = QString("CREATE TABLE IF NOT EXISTS %1("
89 "login VARCHAR(20) NOT NULL PRIMARY KEY, "
90 "%2, "
91 "FOREIGN KEY(login) REFERENCES User(login))")
92 .arg(tableName).arg(columns);
93 if (!query.exec(sql)) {
94 qDebug() << "Failed to create table" << tableName << ":" << query.lastError().text();
95 }
96 }
97 qDebug() << "Statistic tables created (StatisticTask1..4)";
