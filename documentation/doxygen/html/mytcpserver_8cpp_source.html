49void MyTcpServer::slotServerRead()
50{
51 QTcpSocket *clientSocket = qobject_cast<QTcpSocket*>(sender());
52 if (!clientSocket) return;
53
54 long idsock = clientSocket->socketDescriptor();
55
56 while(clientSocket->bytesAvailable() > 0)
57 {
58 QByteArray array = clientSocket->readAll();
59 qDebug() << array << "\n";
60
61 // Раньше здесь было "if(array == \"\\x01\")" — это работало,
62 // только если разделитель приходил ОТДЕЛЬНЫМ, самостоятельным
63 // куском. На практике текст команды и завершающий байт часто
64 // приходят слитно за одно чтение (особенно на localhost),
65 // и такая проверка их не ловила — байт \x01 просто прилипал
66 // к последнему аргументу команды.
67 //
68 // Теперь ищем разделитель ВНУТРИ пришедших байт, а не сравниваем
69 // с ними целиком, и копим неполные сообщения в mBuffers,
70 // пока разделитель не найдётся (в том числе если он придёт
71 // отдельным чтением позже).
72 int sep = array.indexOf('\x01');
73 if (sep == -1) {
74 mBuffers[idsock].append(array);
75 } else {
76 mBuffers[idsock].append(array.left(sep));
77 QString command = QString::fromUtf8(mBuffers[idsock]).trimmed();
78 mBuffers[idsock].clear();
79
80 if (!command.isEmpty()) {
81 UserSession &session = mSessions[idsock];
82 QString response = processCommand(command, session);
83 clientSocket->write(response.toUtf8());
84 }
85
86 // Если после разделителя в этом же чтении есть ещё байты
87 // (например, клиент отправил два сообщения подряд одним
88 // пакетом) — они остаются в буфере для следующего прохода.
89 if (sep + 1 < array.size())
90 mBuffers[idsock].append(array.mid(sep + 1));
91 }
92 }
93}
