7QString processCommand(const QString& command, UserSession& session)
8{
9 QStringList parts = command.split(" ", Qt::SkipEmptyParts);
10 if (parts.isEmpty()) return "";
11
12 QString action = parts[0].toLower();
13
14 // Если есть активное задание и пользователь ввёл одно число,
15 // это считается ответом на задание, а не новой командой.
16 if (session.currentTaskType != 0) {
17 bool ok;
18 int answer = command.trimmed().toInt(&ok);
19 if (ok && parts.size() == 1) {
20 return cmd_answer(answer, session);
21 }
22 // Иначе отменяем текущее задание и обрабатываем как новую команду
23 session.currentTaskType = 0;
24 }
25
26 if (action == "reg") return cmd_reg(parts);
27 if (action == "auth") return cmd_auth(parts, session);
28 if (action == "logout") return cmd_logout(session);
29 if (action == "whoami") return cmd_whoami(session);
30 if (action == "del") return cmd_del(parts, session);
31 if (action == "role") return cmd_role(parts, session);
32 if (action == "users") return cmd_users(session);
33 if (action == "stats") return cmd_stats(session);
34 if (action == "help") return cmd_help();
35
36 // Команды для выбора задания: "task1 5", "task2 3" и т.д.
37 if (action == "task1") {
38 if (parts.size() < 2) return "error: usage: task1 <1-40>\r\n";
39 bool ok;
40 int num = parts[1].toInt(&ok);
41 if (!ok || num < 1 || num > 40) return "error: task number must be 1-40\r\n";
42 return cmd_task(1, num, session);
43 }
44 if (action == "task2") {
45 if (parts.size() < 2) return "error: usage: task2 <1-10>\r\n";
46 bool ok;
47 int num = parts[1].toInt(&ok);
48 if (!ok || num < 1 || num > 10) return "error: task number must be 1-10\r\n";
49 return cmd_task(2, num, session);
50 }
51 if (action == "task3") {
52 if (parts.size() < 2) return "error: usage: task3 <1-10>\r\n";
53 bool ok;
54 int num = parts[1].toInt(&ok);
55 if (!ok || num < 1 || num > 10) return "error: task number must be 1-10\r\n";
56 return cmd_task(3, num, session);
57 }
58 if (action == "task4") {
59 if (parts.size() < 2) return "error: usage: task4 <1-40>\r\n";
60 bool ok;
61 int num = parts[1].toInt(&ok);
62 if (!ok || num < 1 || num > 40) return "error: task number must be 1-40\r\n";
63 return cmd_task(4, num, session);
64 }
65
66 // Команда просмотра личной статистики: "mystats" или "mystats 1"
67 if (action == "mystats") return cmd_mystats(parts, session);
68
69 // Echo для неизвестных команд
70 return command + "\r\n";
71}

73QString cmd_reg(const QStringList& parts)
74{
75 // reg login password email [role] - роль последняя
76 if (parts.size() < 4) {
77 return "error: usage: reg login password email [role]\r\n";
78 }
79
80 QString login = parts[1];
81 QString password = parts[2];
82 QString email = parts[3];
83 UserRole role = ROLE_USER;
84
85 if (parts.size() >= 5) {
86 QString roleStr = parts[4].toLower();
87 if (roleStr != "user" && roleStr != "moderator" && roleStr != "admin") {
88 return "error: invalid role (use: user, moderator, admin)\r\n";
89 }
90 role = stringToRole(roleStr);
91 }
92
93 if (DatabaseManager::getInstance()->userExists(login)) {
94 return "error: user exists\r\n";
95 }
96
97 if (DatabaseManager::getInstance()->regUser(login, password, email, role)) {
98 // При успешной регистрации создаём строки статистики
99 // во всех 4 таблицах (StatisticTask1..4) с нулевыми значениями
100 DatabaseManager::getInstance()->createStatisticRows(login);
101 return "reg+\r\n";
102 }
103 return "error: reg failed\r\n";
104}

291QString cmd_task(int taskType, int taskNum, UserSession& session)
292{
293 // Требуется авторизация
294 if (session.login.isEmpty()) {
295 return "error: login required to use tasks\r\n";
296 }
297
298 QString problemText;
299 int correctAnswer = 0;
300
301 // В зависимости от типа задания вызываем соответствующую функцию генерации
302 switch (taskType) {
303 case 1: {
304 Task1Problem p = task1_generate(taskNum);
305 problemText = p.text;
306 correctAnswer = p.correctAnswer;
307 break;
308 }
309 case 2: {
310 Task2Problem p = task2_generate(taskNum);
311 problemText = p.text;
312 correctAnswer = p.correctAnswer;
313 break;
314 }
315 case 3: {
316 Task3Problem p = task3_generate(taskNum);
317 problemText = p.text;
318 correctAnswer = p.correctAnswer;
319 break;
320 }
321 case 4: {
322 Task4Problem p = task4_generate(taskNum);
323 problemText = p.text;
324 correctAnswer = p.correctAnswer;
325 break;
326 }
327 default:
328 return "error: invalid task type (1-4)\r\n";
329 }
330
331 // Сохраняем состояние активного задания в сессии
332 session.currentTaskType = taskType;
333 session.currentTaskNum = taskNum;
334 session.currentCorrectAnswer = correctAnswer;
335
336 return problemText;
337}

339// Обработка ответа пользователя на активное задание.
340// Проверяет ответ, обновляет статистику в БД (+1 или -1),
341// отправляет результат и сбрасывает активное задание.
342QString cmd_answer(int userAnswer, UserSession& session)
343{
344 // Проверяем, есть ли активное задание
345 if (session.currentTaskType == 0) {
346 return "error: no active task. Use task1/4 <1-40> or task2/3 <1-10> first.\r\n";
347 }
348
349 int taskType = session.currentTaskType;
350 int taskNum = session.currentTaskNum;
351 int correctAnswer = session.currentCorrectAnswer;
352
353 // Определяем правильность ответа
354 bool isCorrect = (userAnswer == correctAnswer);
355
356 // Обновляем статистику в БД: +1 за правильный, -1 за неправильный
357 int delta = isCorrect ? 1 : -1;
358 DatabaseManager::getInstance()->updateStatistic(session.login, taskType, taskNum, delta);
359
360 // Получаем текущее значение из таблицы для отображения
361 QVector<int> row = DatabaseManager::getInstance()->getStatisticRow(session.login, taskType);
362 int currentValue = row.value(taskNum - 1, 0); // индекс 0..39 для заданий 1..40
363
364 // Формируем строку с именем таблицы для вывода
365 QString tableName = QString("StatisticTask%1[%2]").arg(taskType).arg(taskNum);
366
367 // Сбрасываем активное задание
368 session.currentTaskType = 0;
369 session.currentTaskNum = 0;
370 session.currentCorrectAnswer = 0;
371
372 if (isCorrect) {
373 return QString("Верно! %1 += 1 (текущее значение: %2)\r\n")
374 .arg(tableName).arg(currentValue);
375 } else {
376 return QString("Неверно! %1 -= 1 (текущее: %2). "
377 "Правильный ответ: %3\r\n")
378 .arg(tableName).arg(currentValue).arg(correctAnswer);
379 }
380}

382// Просмотр личной статистики по заданиям.
383// "mystats" — показывает все 4 Task, "mystats 1" — только Task1 и т.д.
384QString cmd_mystats(const QStringList& parts, const UserSession& session)
385{
386 // Требуется авторизация
387 if (session.login.isEmpty()) {
388 return "error: login required\r\n";
389 }
390
391 // Определяем, какой Task запрошен
392 int requestedTask = 0;
393 if (parts.size() >= 2) {
394 int t = parts[1].toInt();
395 if (t < 0 || t > 4) {
396 return "error: usage: mystats [0-4]\r\n";
397 }
398 requestedTask = t;
399 }
400
401 QString result = QString("=== Статистика пользователя: %1 ===\r\n").arg(session.login);
402
403 // Определяем диапазон Task для отображения
404 int startTask = (requestedTask == 0) ? 1 : requestedTask;
405 int endTask = (requestedTask == 0) ? 4 : requestedTask;
406
407 for (int taskType = startTask; taskType <= endTask; taskType++) {
408 QString taskName = QString("Task%1").arg(taskType);
409 result += QString("--- %1 ---\r\n").arg(taskName);
410
411 QVector<int> row = DatabaseManager::getInstance()->getStatisticRow(session.login, taskType);
412
413 // У task2 и task3 по 10 шаблонов, у task1 и task4 — по 40.
414 int cells = (taskType == 2 || taskType == 3) ? 10 : 40;
415
416 // Выводим значения по 10 в строке для компактности
417 for (int line = 0; line < cells / 10; line++) {
418 QStringList vals;
419 for (int col = 0; col < 10; col++) {
420 int idx = line * 10 + col;
421 vals << QString("%1:%2").arg(idx + 1).arg(row[idx]);
422 }
423 result += " " + vals.join(" ") + "\r\n";
424 }
425 }
426
427 return result;
428}
