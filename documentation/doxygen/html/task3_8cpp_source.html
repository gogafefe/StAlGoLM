25// Выполняет классический метод половинного деления. До вызова должно быть
26// выполнено условие f(a) * f(b) <= 0.
27static BisectionResult solveBisection(const std::function<double(double)>& f,
28 double a, double b, double epsilon)
29{
30 double fa = f(a);
31 int iterations = 0;
32
33 while ((b - a) / 2.0 > epsilon && iterations < 200) {
34 double middle = (a + b) / 2.0;
35 double fm = f(middle);
36 iterations++;
37
38 if (std::fabs(fm) < 1e-12) {
39 return {middle, iterations};
40 }
41
42 if (fa * fm <= 0.0) {
43 b = middle;
44 } else {
45 a = middle;
46 fa = fm;
47 }
48 }
49
50 return {(a + b) / 2.0, iterations};
51}
